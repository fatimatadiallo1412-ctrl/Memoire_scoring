def enum_col(enum_cls, name, **kw):
    """Colonne à valeurs fermées : stockée en texte, avec une contrainte CHECK en base."""
    return mapped_column(
        SAEnum(enum_cls, name=name, native_enum=False, length=32,
               create_constraint=True, validate_strings=True,
               values_callable=lambda e: [m.value for m in e]),
        **kw)

from datetime import date, datetime, timezone

from fastapi import APIRouter, Depends, HTTPException, Query, status
from sqlalchemy import select, text
from sqlalchemy.orm import Session

from app.db import get_session
from app.deps import exiger_roles
from app.models import Client, DemandeCredit, Journal, ProduitCredit, Score, Utilisateur
from app.models.enums import Role, StatutDossier

router = APIRouter(prefix="/demandes", tags=["demandes"])

SAISIE = (Role.AGENT_CREDIT, Role.ADMINISTRATEUR)
LECTURE = (Role.AGENT_CREDIT, Role.DECIDEUR, Role.ADMINISTRATEUR)
MODIFIABLES = (StatutDossier.BROUILLON, StatutDossier.COMPLET,
               StatutDossier.SCORE, StatutDossier.INSTRUIT)


def verifier_produit(session: Session, produit_id: int, montant: int, duree: int) -> None:
    produit = session.get(ProduitCredit, produit_id)
    if produit is None or not produit.actif:
        raise HTTPException(status.HTTP_422_UNPROCESSABLE_ENTITY, "Produit inconnu ou inactif")
    if not produit.montant_min <= montant <= produit.montant_max:
        raise HTTPException(
            status.HTTP_422_UNPROCESSABLE_ENTITY,
            f"Montant hors limites du produit ({produit.montant_min} à {produit.montant_max} FCFA)")
    if not produit.duree_min_mois <= duree <= produit.duree_max_mois:
        raise HTTPException(
            status.HTTP_422_UNPROCESSABLE_ENTITY,
            f"Durée hors limites du produit ({produit.duree_min_mois} à {produit.duree_max_mois} mois)")


def charger_demande(session: Session, demande_id: int) -> DemandeCredit:
    demande = session.get(DemandeCredit, demande_id)
    if demande is None:
        raise HTTPException(status.HTTP_404_NOT_FOUND, "Dossier introuvable")
    return demande


@router.post("", response_model=DemandeOut, status_code=status.HTTP_201_CREATED)
def creer_demande(donnees: DemandeIn, session: Session = Depends(get_session),
                  utilisateur: Utilisateur = Depends(exiger_roles(*SAISIE))):
    if session.get(Client, donnees.client_id) is None:
        raise HTTPException(status.HTTP_422_UNPROCESSABLE_ENTITY, "Client inconnu")
    verifier_produit(session, donnees.produit_id, donnees.montant_demande, donnees.duree_mois)

    prochain_id = session.scalar(text("SELECT nextval(pg_get_serial_sequence('ops.demande_credit', 'id'))"))
    jour = donnees.date_demande or date.today()
    demande = DemandeCredit(
        id=prochain_id,
        numero=f"DC{jour.year}-{prochain_id:06d}",
        client_id=donnees.client_id,
        produit_id=donnees.produit_id,
        instructeur_id=utilisateur.id,
        montant_demande=donnees.montant_demande,
        duree_mois=donnees.duree_mois,
        objet=donnees.objet,
        date_demande=jour,
    )
    session.add(demande)
    session.add(Journal(utilisateur_id=utilisateur.id, action="dossier_cree",
                        entite="demande_credit", entite_id=prochain_id))
    session.commit()
    session.refresh(demande)
    return demande


@router.get("", response_model=list[DemandeOut])
def lister_demandes(
    statut: StatutDossier | None = None,
    client_id: int | None = None,
    limite: int = Query(default=20, ge=1, le=100),
    session: Session = Depends(get_session),
    utilisateur: Utilisateur = Depends(exiger_roles(*LECTURE)),
):
    requete = select(DemandeCredit).order_by(DemandeCredit.id.desc()).limit(limite)
    if statut:
        requete = requete.where(DemandeCredit.statut == statut)
    if client_id:
        requete = requete.where(DemandeCredit.client_id == client_id)
    return session.scalars(requete).all()


@router.get("/{demande_id}", response_model=DemandeOut)
def lire_demande(demande_id: int, session: Session = Depends(get_session),
                 utilisateur: Utilisateur = Depends(exiger_roles(*LECTURE))):
    return charger_demande(session, demande_id)


@router.put("/{demande_id}", response_model=DemandeOut)
def modifier_demande(demande_id: int, donnees: DemandeIn, session: Session = Depends(get_session),
                     utilisateur: Utilisateur = Depends(exiger_roles(*SAISIE))):
    demande = charger_demande(session, demande_id)
    if demande.statut not in MODIFIABLES:
        raise HTTPException(status.HTTP_409_CONFLICT,
                            f"Un dossier au statut {demande.statut.value} ne peut plus être modifié")
    verifier_produit(session, donnees.produit_id, donnees.montant_demande, donnees.duree_mois)

    demande.produit_id = donnees.produit_id
    demande.montant_demande = donnees.montant_demande
    demande.duree_mois = donnees.duree_mois
    demande.objet = donnees.objet
    demande.donnees_modifiees_le = text("now()")   # le score éventuel devient périmé
    if demande.statut in (StatutDossier.SCORE, StatutDossier.INSTRUIT):
        changer_statut(demande, StatutDossier.COMPLET)

    session.add(Journal(utilisateur_id=utilisateur.id, action="dossier_modifie",
                        entite="demande_credit", entite_id=demande_id))
    session.commit()
    session.refresh(demande)
    return demande


@router.post("/{demande_id}/completer", response_model=DemandeOut)
def completer_demande(demande_id: int, session: Session = Depends(get_session),
                      utilisateur: Utilisateur = Depends(exiger_roles(*SAISIE))):
    demande = charger_demande(session, demande_id)
    client = session.get(Client, demande.client_id)

    manquants = []
    if client.revenu_mensuel <= 0:
        manquants.append("revenu mensuel du client")
    if manquants:
        raise HTTPException(status.HTTP_422_UNPROCESSABLE_ENTITY,
                            "Dossier incomplet : " + ", ".join(manquants))

    changer_statut(demande, StatutDossier.COMPLET)
    session.add(Journal(utilisateur_id=utilisateur.id, action="dossier_complet",
                        entite="demande_credit", entite_id=demande_id))
    session.commit()
    session.refresh(demande)
    return demande
@router.post("/{demande_id}/scorer", response_model=ScoreOut)
def scorer_demande(demande_id: int, session: Session = Depends(get_session),
                   utilisateur: Utilisateur = Depends(exiger_roles(*SAISIE))):
    demande = charger_demande(session, demande_id)
    if demande.statut not in (StatutDossier.COMPLET, StatutDossier.SCORE):
        raise HTTPException(status.HTTP_409_CONFLICT,
                            f"Un dossier au statut {demande.statut.value} ne peut pas être scoré")
    client = session.get(Client, demande.client_id)
    produit = session.get(ProduitCredit, demande.produit_id)
    if client.revenu_mensuel <= 0:
        raise HTTPException(status.HTTP_422_UNPROCESSABLE_ENTITY,
                            "Le revenu mensuel du client est nécessaire au scoring")

    parametres = lire_parametres(session)
    reco, variables = evaluer_dossier(
        client, demande, produit, parametres,
        antecedent=niveau_antecedent(session, client.id, demande.date_demande))

    if reco["decision"] == "accord":
        montant_suggere, p_suggere = demande.montant_demande, reco["p_defaut"]
    else:
        montant_suggere, p_suggere = reco["montant"], reco["p_defaut"]

    score = Score(
        demande_id=demande.id,
        version_modele=VERSION_MODELE,
        p_defaut=reco["p_demande"],
        seuil_applique=reco["seuil_applique"],
        recommandation=reco["decision"],
        montant_suggere=montant_suggere,
        p_montant_suggere=p_suggere,
        variables_entree=variables,
        facteurs={"motifs": reco["motifs"],
                  "endettement_demande": round(reco["endettement_demande"], 3),
                  "parametres": parametres},
    )
    session.add(score)

                        entite="demande_credit", entite_id=demande_id,
                        details={"recommandation": reco["decision"],
                                 "p_defaut": round(reco["p_demande"], 4)}))
    session.commit()
    session.refresh(score)
    return score

@router.post("/demandes/{demande_id}/decaisser", response_model=CreditOut,
             status_code=status.HTTP_201_CREATED)
def decaisser(demande_id: int, donnees: DecaissementIn | None = None,
              session: Session = Depends(get_session),
              utilisateur: Utilisateur = Depends(
                  exiger_roles(Role.AGENT_CREDIT, Role.ADMINISTRATEUR))):
    demande = charger_demande(session, demande_id)
    if demande.statut not in (StatutDossier.ACCORDE, StatutDossier.ACCORDE_MODIFIE):
        raise HTTPException(status.HTTP_409_CONFLICT,
                            "Seul un dossier accordé peut être décaissé")

    # Les conditions du crédit sont celles de la dernière décision du directeur
    decision = session.scalars(
        select(Decision).where(Decision.demande_id == demande_id)
        .order_by(Decision.decide_le.desc()).limit(1)).first()
    if decision is None or decision.montant_accorde is None:
        raise HTTPException(status.HTTP_409_CONFLICT,
                            "Aucune décision d'accord trouvée pour ce dossier")

    produit = session.get(ProduitCredit, demande.produit_id)
    taux = float(produit.taux_annuel) / 100          # 24.00 (%) -> 0.24
    montant, duree = decision.montant_accorde, decision.duree_accordee
    jour = (donnees.date_decaissement if donnees and donnees.date_decaissement
            else date.today())

    credit = Credit(
        demande_id=demande.id, client_id=demande.client_id,
        montant_accorde=montant, duree_mois=duree, taux_annuel=taux,
        mensualite=mensualite_constante(montant, duree, taux),
        date_decaissement=jour, decaisse_par=utilisateur.id)
    session.add(credit)
    session.flush()          # attribue credit.id sans valider la transaction

    for ligne in generer_calendrier(montant, duree, taux, jour):
        session.add(Echeance(credit_id=credit.id, **ligne))

    changer_statut(demande, StatutDossier.DECAISSE)
    changer_statut(demande, StatutDossier.EN_REMBOURSEMENT)
    session.add(Journal(utilisateur_id=utilisateur.id, action="decaissement",
                        entite="credit", entite_id=credit.id,
                        details={"demande_id": demande.id, "montant": montant,
                                 "duree_mois": duree}))
    session.commit()
    session.refresh(credit)
    return credit

def _situation(session: Session, credit: Credit) -> dict:
    echeances = session.scalars(select(Echeance).where(Echeance.credit_id == credit.id)).all()
    total_paye = session.scalar(
        select(func.coalesce(func.sum(Paiement.montant), 0))
        .where(Paiement.credit_id == credit.id))
    return situation_credit(echeances, int(total_paye), date.today())


@router.post("/credits/{credit_id}/paiements", response_model=PaiementOut,
             status_code=status.HTTP_201_CREATED)
def enregistrer_paiement(credit_id: int, donnees: PaiementIn,
                         session: Session = Depends(get_session),
                         utilisateur: Utilisateur = Depends(
                             exiger_roles(Role.AGENT_CREDIT, Role.ADMINISTRATEUR))):
    credit = _credit_ou_404(session, credit_id)
    demande = session.get(DemandeCredit, credit.demande_id)
    if demande.statut not in (StatutDossier.EN_REMBOURSEMENT, StatutDossier.CONTENTIEUX):
        raise HTTPException(status.HTTP_409_CONFLICT,
                            "Ce crédit n'accepte plus de paiement")

    jour = donnees.date_paiement or date.today()
    if jour > date.today():
        raise HTTPException(status.HTTP_422_UNPROCESSABLE_ENTITY,
                            "La date de paiement ne peut pas être dans le futur")
    if jour < credit.date_decaissement:
        raise HTTPException(status.HTTP_422_UNPROCESSABLE_ENTITY,
                            "La date de paiement précède le décaissement")

    situation = _situation(session, credit)
    if donnees.montant > situation["reste_a_payer"]:
        raise HTTPException(
            status.HTTP_422_UNPROCESSABLE_ENTITY,
            f"Le montant dépasse le reste à payer ({situation['reste_a_payer']})")

    paiement = Paiement(credit_id=credit.id, montant=donnees.montant,
                        date_paiement=jour, mode=donnees.mode,
                        enregistre_par=utilisateur.id)
    session.add(paiement)
    session.flush()

    # Si ce paiement solde le crédit, le dossier passe à "solde"
    if donnees.montant == situation["reste_a_payer"]:
        changer_statut(demande, StatutDossier.SOLDE)

    session.add(Journal(utilisateur_id=utilisateur.id, action="paiement",
                        entite="credit", entite_id=credit.id,
                        details={"montant": donnees.montant, "mode": donnees.mode}))
    session.commit()
    session.refresh(paiement)
    return paiement

    "accord_reduit": TypeDecision.ACCORDER_MODIFIE,

    TypeDecision.ACCORDER_MODIFIE: StatutDossier.ACCORDE_MODIFIE,

@router.post("/{demande_id}/decider", response_model=DecisionOut,
             status_code=status.HTTP_201_CREATED)
def decider(demande_id: int, donnees: DecisionIn, session: Session = Depends(get_session),
            utilisateur: Utilisateur = Depends(exiger_roles(Role.DECIDEUR))):
    demande = charger_demande(session, demande_id)
    if demande.statut != StatutDossier.SOUMIS:
        raise HTTPException(status.HTTP_409_CONFLICT,
                            "Seul un dossier soumis à décision peut faire l'objet d'une décision")
    score = exiger_score_valide(session, demande)

    choix = donnees.decision
    montant = duree = None
    if choix == TypeDecision.ACCORDER:
        montant, duree = demande.montant_demande, demande.duree_mois
    elif choix == TypeDecision.ACCORDER_MODIFIE:
        if donnees.montant_accorde is None or donnees.duree_accordee is None:
            raise HTTPException(status.HTTP_422_UNPROCESSABLE_ENTITY,
                                "Montant et durée accordés obligatoires pour un accord avec modification")
        montant, duree = donnees.montant_accorde, donnees.duree_accordee
        verifier_produit(session, demande.produit_id, montant, duree)

    concordance = choix == EQUIVALENT_MODELE[score.recommandation]
    if not concordance and not (donnees.commentaire and donnees.commentaire.strip()):
        raise HTTPException(
            status.HTTP_422_UNPROCESSABLE_ENTITY,
            "Un motif est obligatoire quand la décision diffère de la recommandation du modèle")

    decision = Decision(
        demande_id=demande.id, score_id=score.id, decideur_id=utilisateur.id,
        decision=choix, montant_accorde=montant, duree_accordee=duree,
        commentaire=donnees.commentaire, recommandation_modele=score.recommandation,
        concordance=concordance)
    session.add(decision)
    changer_statut(demande, STATUT_APRES[choix])
    session.add(Journal(utilisateur_id=utilisateur.id, action="decision",
                        entite="demande_credit", entite_id=demande_id,
                        details={"decision": choix.value, "concordance": concordance}))
    session.commit()
    session.refresh(decision)
    return decision

"""Crée un compte de connexion (rôle client) rattaché à une fiche client existante.

Usage :
    python -m app.creer_compte_client --client-id 3 --email awa.niass@client.test --nom "Awa Niass"
Le mot de passe est demandé en saisie masquée (il n'apparaît pas dans l'historique).
"""
import argparse
import getpass
import sys

from sqlalchemy import select

from app.db import SessionLocal
from app.models import Client, Utilisateur
from app.models.enums import Role
from app.security import hasher_mot_de_passe


def main() -> None:
    parseur = argparse.ArgumentParser(description=__doc__)
    parseur.add_argument("--client-id", type=int, required=True)
    parseur.add_argument("--email", required=True)
    parseur.add_argument("--nom", required=True, help="Nom complet affiché")
    args = parseur.parse_args()

    mot_de_passe = getpass.getpass("Mot de passe du compte client : ")
    if len(mot_de_passe) < 8:
        sys.exit("Mot de passe trop court (8 caractères minimum).")

    with SessionLocal() as session:
        if session.get(Client, args.client_id) is None:
            sys.exit(f"Aucun client avec l'id {args.client_id}.")
        if session.scalars(select(Utilisateur).where(
                Utilisateur.client_id == args.client_id)).first():
            sys.exit("Ce client a déjà un compte.")
        if session.scalars(select(Utilisateur).where(
                Utilisateur.email == args.email)).first():
            sys.exit("Cet email est déjà utilisé.")

        session.add(Utilisateur(
            email=args.email, nom_complet=args.nom,
            mot_de_passe_hash=hasher_mot_de_passe(mot_de_passe),
            role=Role.CLIENT, client_id=args.client_id))
        session.commit()
        print(f"Compte créé : {args.email} (client {args.client_id})")


if __name__ == "__main__":
    main()
