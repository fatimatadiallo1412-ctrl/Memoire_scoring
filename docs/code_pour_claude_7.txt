import { useState } from "react";
import { Navigate, useNavigate } from "react-router-dom";
import { useAuth } from "./auth.jsx";
import { ACCUEIL } from "./roles.js";

export default function Connexion() {
  const { utilisateur, connecter } = useAuth();
  const navigate = useNavigate();
  const [email, setEmail] = useState("");
  const [motDePasse, setMotDePasse] = useState("");
  const [erreur, setErreur] = useState("");
  const [envoi, setEnvoi] = useState(false);

  if (utilisateur) return <Navigate to={ACCUEIL[utilisateur.role]} replace />;

  async function soumettre(evenement) {
    evenement.preventDefault();
    setErreur("");
    setEnvoi(true);
    try {
      const moi = await connecter(email.trim(), motDePasse);
      navigate(ACCUEIL[moi.role], { replace: true });
    } catch (e) {
      setErreur(e.statut === 401 ? "E-mail ou mot de passe incorrect." : e.message);
    } finally {
      setEnvoi(false);
    }
  }

  return (
    <div className="page-connexion">
      <form className="carte" onSubmit={soumettre}>
        <h1>Connexion</h1>
        <label>
          E-mail
          <input type="email" value={email} onChange={(e) => setEmail(e.target.value)} required autoFocus />
        </label>
        <label>
          Mot de passe
          <input type="password" value={motDePasse} onChange={(e) => setMotDePasse(e.target.value)} required />
        </label>
        {erreur && <p className="erreur">{erreur}</p>}
        <button type="submit" disabled={envoi}>{envoi ? "Connexion…" : "Se connecter"}</button>
      </form>
    </div>
  );
}

export default function Clients() {
  const { utilisateur } = useAuth();
  const peutCreer = ["agent_credit", "administrateur"].includes(utilisateur.role);
  const [q, setQ] = useState("");
  const [clients, setClients] = useState([]);
  const [erreur, setErreur] = useState("");
  const [chargement, setChargement] = useState(true);

  useEffect(() => {
    const terme = q.trim();
    if (terme.length === 1) return; // l'API exige au moins 2 caractères
    let annule = false;
    const minuteur = setTimeout(async () => {
      setChargement(true);
      setErreur("");
      try {
        const liste = await appeler("/clients", { params: { q: terme, limite: 50 } });
        if (!annule) setClients(liste);
      } catch (e) {
        if (!annule) setErreur(e.message);
      } finally {
        if (!annule) setChargement(false);
      }
    }, 300);
    return () => {
      annule = true;
      clearTimeout(minuteur);
    };
  }, [q]);

  return (
    <Cadre titre="Clients">
      {peutCreer && (
        <p><Link className="lien-bouton" to="/clients/nouveau">+ Nouveau client</Link></p>
      )}
      <input
        className="recherche"
        type="search"
        placeholder="Nom, prénom, téléphone ou code client (2 caractères minimum)"
        value={q}
        onChange={(e) => setQ(e.target.value)}
      />
      {erreur && <p className="erreur">{erreur}</p>}
      {chargement && <p className="discret">Recherche…</p>}
      {!chargement && !erreur && (
        <p className="discret">
          {clients.length} client(s). Plusieurs clients peuvent porter le même nom : repérez le bon par
          la date de naissance, le téléphone ou le code client.
        </p>
      )}
      <table className="tableau">
        <thead>
          <tr>
            <th>Code</th><th>Nom</th><th>Prénom</th><th>Naissance</th><th>Téléphone</th><th>Adresse</th><th></th>
          </tr>
        </thead>
        <tbody>
          {clients.map((c) => (
            <tr key={c.id}>
              <td>{c.code_client}</td>
              <td>{c.nom}</td>
              <td>{c.prenom}</td>
              <td>{dateFr(c.date_naissance)}</td>
              <td>{c.telephone}</td>
              <td>{c.adresse}</td>
              <td><Link to={"/clients/" + c.id}>Ouvrir</Link></td>
            </tr>
          ))}
        </tbody>
      </table>
    </Cadre>
  );
}

// Champs d'identité : les modifier n'invalide pas les scores existants.
const IDENTITE = ["nom", "prenom", "telephone", "adresse", "sexe"];
const ENTIERS = ["anciennete_activite_mois", "nb_enfants", "taille_menage", "revenu_mensuel"];
const FACULTATIFS = ["sexe", "telephone", "adresse", "profession"];

function libelleMetier(code, metiers) {
  if (!code) return "";
  const metier = metiers.find((m) => m.code === code);
  return metier ? metier.libelle : code;
}

function versFormulaire(c, metiers) {
  return {
    nom: c.nom,
    prenom: c.prenom,
    date_naissance: c.date_naissance,
    sexe: c.sexe ?? "",
    telephone: c.telephone ?? "",
    adresse: c.adresse ?? "",
    type_revenu: c.type_revenu,
    profession: libelleMetier(c.profession, metiers),
    situation_familiale: c.situation_familiale,
    type_logement: c.type_logement,
    anciennete_activite_mois: String(c.anciennete_activite_mois),
    nb_enfants: String(c.nb_enfants),
    taille_menage: String(c.taille_menage),
    possede_bien_immobilier: c.possede_bien_immobilier,
    possede_vehicule: c.possede_vehicule,
    revenu_mensuel: String(c.revenu_mensuel),
  };
}

function valeurApi(cle, valeur) {
  if (ENTIERS.includes(cle)) return parseInt(valeur, 10);
  if (FACULTATIFS.includes(cle)) {
    const propre = typeof valeur === "string" ? valeur.trim() : valeur;
    return propre === "" ? null : propre;
  }
  return valeur;
}

function Liste({ valeur, options, onChange, facultatif }) {
  return (
    <select value={valeur} onChange={(e) => onChange(e.target.value)} required={!facultatif}>
      {facultatif && <option value="">(non renseigné)</option>}
      {options.map(([cle, libelle]) => (
        <option key={cle} value={cle}>{libelle}</option>
      ))}
    </select>
  );
}

export default function FicheClient() {
  const { id } = useParams();
  const { utilisateur } = useAuth();
  const peutDemander = ["agent_credit", "administrateur"].includes(utilisateur.role);
  const responsable = ["decideur", "administrateur"].includes(utilisateur.role);
  const [client, setClient] = useState(null);
  const [metiers, setMetiers] = useState([]);
  const [f, setF] = useState(null);
  const [erreur, setErreur] = useState("");
  const [message, setMessage] = useState("");
  const [envoi, setEnvoi] = useState(false);

  useEffect(() => {
    setClient(null);
    setErreur("");
    Promise.all([appeler("/clients/" + id), appeler("/professions")])
      .then(([c, liste]) => {
        setMetiers(liste);
        setClient(c);
        setF(versFormulaire(c, liste));
      })
      .catch((e) => setErreur(e.message));
  }, [id]);

  function changer(cle, valeur) {
    setMessage("");
    setF((ancien) => ({ ...ancien, [cle]: valeur }));
  }

  async function enregistrer(evenement) {
    evenement.preventDefault();
    setErreur("");
    setMessage("");
    const original = versFormulaire(client, metiers);
    const modifies = Object.keys(f).filter((cle) => f[cle] !== original[cle]);
    if (modifies.length === 0) {
      setMessage("Aucune modification à enregistrer.");
      return;
    }
    const corps = Object.fromEntries(modifies.map((cle) => [cle, valeurApi(cle, f[cle])]));
    setEnvoi(true);
    try {
      const maj = await appeler("/clients/" + id, { methode: "PATCH", json: corps });
      const liste = await appeler("/professions");
      invaliderMetiers();
      setMetiers(liste);
      setClient(maj);
      setF(versFormulaire(maj, liste));
      setMessage("Enregistré : " + modifies.join(", ") + ".");
    } catch (e) {
      setErreur(e.message);
    } finally {
      setEnvoi(false);
    }
  }

  if (erreur && !client) {
    return (
      <Cadre titre="Client">
        <p className="erreur">{erreur}</p>
        <Link to="/clients">Retour à la liste</Link>
      </Cadre>
    );
  }
  if (!client || !f) {
    return (
      <Cadre titre="Client">
        <p className="discret">Chargement…</p>
      </Cadre>
    );
  }

  const marque = (cle) => (IDENTITE.includes(cle) ? "" : " ⚠");
  const aClasser = client.profession && !client.categorie_profession;

  return (
    <Cadre titre={client.prenom + " " + client.nom}>
      <p className="discret">
        Code {client.code_client} · gestionnaire n° {client.gestionnaire_id ?? "aucun"} · créé le{" "}
        {dateFr(client.cree_le.slice(0, 10))}
      </p>
      {client.categorie_profession && (
        <p className="discret">
          Catégorie de métier utilisée par le modèle :{" "}
          {LIBELLE_CATEGORIE[client.categorie_profession] || client.categorie_profession}
        </p>
      )}
      {aClasser && (
        <p className="avertissement">
          Le métier « {libelleMetier(client.profession, metiers)} » n'est pas encore classé : tant
          qu'un responsable ne lui a pas attribué une catégorie, le score utilise « Non renseigné ».{" "}
          {responsable
            ? <Link to="/metiers">Classer les métiers</Link>
            : "Le directeur ou l'administrateur doit le classer."}
        </p>
      )}
      <p className="discret">
        ⚠ : modifier ce champ rend non valides les scores déjà calculés pour ce client ; il faudra
        recalculer le score des dossiers concernés.
      </p>
      {peutDemander && (
        <p><Link className="lien-bouton" to={"/demandes/nouvelle?client_id=" + client.id}>+ Nouvelle demande de crédit</Link></p>
      )}
      <form className="formulaire" onSubmit={enregistrer}>
        <label>Nom
          <input value={f.nom} onChange={(e) => changer("nom", e.target.value)} required maxLength={100} />
        </label>
        <label>Prénom
          <input value={f.prenom} onChange={(e) => changer("prenom", e.target.value)} required maxLength={100} />
        </label>
        <label>{"Date de naissance" + marque("date_naissance")}
          <input type="date" value={f.date_naissance} max={new Date().toISOString().slice(0, 10)}
            onChange={(e) => changer("date_naissance", e.target.value)} required />
        </label>
        <label>Sexe
          <Liste valeur={f.sexe} onChange={(v) => changer("sexe", v)} facultatif
            options={[["F", "Féminin"], ["M", "Masculin"]]} />
        </label>
        <label>Téléphone
          <input type="tel" value={f.telephone} onChange={(e) => changer("telephone", e.target.value)} maxLength={20} />
        </label>
        <label>Adresse
          <input value={f.adresse} onChange={(e) => changer("adresse", e.target.value)} maxLength={200} />
        </label>
        <label>{"Type de revenu" + marque("type_revenu")}
          <Liste valeur={f.type_revenu} options={TYPES_REVENU} onChange={(v) => changer("type_revenu", v)} />
        </label>
        <label>{"Profession" + marque("profession")}
          <ChampMetier valeur={f.profession} onChange={(v) => changer("profession", v)} metiers={metiers} />
        </label>
        <label>{"Situation familiale" + marque("situation_familiale")}
          <Liste valeur={f.situation_familiale} options={SITUATIONS} onChange={(v) => changer("situation_familiale", v)} />
        </label>
        <label>{"Logement" + marque("type_logement")}
          <Liste valeur={f.type_logement} options={LOGEMENTS} onChange={(v) => changer("type_logement", v)} />
        </label>
        <label>{"Ancienneté de l'activité (mois)" + marque("anciennete_activite_mois")}
          <input type="number" min="0" value={f.anciennete_activite_mois}
            onChange={(e) => changer("anciennete_activite_mois", e.target.value)} required />
        </label>
        <label>{"Enfants" + marque("nb_enfants")}
          <input type="number" min="0" max="30" value={f.nb_enfants}
            onChange={(e) => changer("nb_enfants", e.target.value)} required />
        </label>
        <label>{"Taille du ménage" + marque("taille_menage")}
          <input type="number" min="1" max="40" value={f.taille_menage}
            onChange={(e) => changer("taille_menage", e.target.value)} required />
        </label>
        <label>{"Revenu mensuel moyen estimé (FCFA)" + marque("revenu_mensuel")}

            onChange={(e) => changer("revenu_mensuel", e.target.value)} required />

function nomClient(c) {
  return c.prenom + " " + c.nom + " (" + c.code_client + ")";
}

export default function Dossiers() {
  const { utilisateur } = useAuth();
  const [statut, setStatut] = useState(utilisateur.role === "decideur" ? "soumis_decision" : "");
  const [q, setQ] = useState("");
  const [demandes, setDemandes] = useState([]);
  const [noms, setNoms] = useState({});
  const [nbClients, setNbClients] = useState(null);
  const [erreur, setErreur] = useState("");
  const [chargement, setChargement] = useState(true);

  useEffect(() => {
    const terme = q.trim();
    if (terme.length === 1) return; // l'API exige au moins 2 caractères
    let annule = false;
    const minuteur = setTimeout(async () => {
      setChargement(true);
      setErreur("");
      try {
        if (!terme) {
          const liste = await appeler("/demandes", { params: { statut, limite: 50 } });
          if (!annule) {
            setNbClients(null);
            setDemandes(liste);
          }
        } else {
          const clients = await appeler("/clients", { params: { q: terme, limite: 20 } });
          const listes = await Promise.all(
            clients.map((c) => appeler("/demandes", { params: { client_id: c.id, statut, limite: 50 } }))
          );
          if (!annule) {
            setNoms((a) => ({ ...a, ...Object.fromEntries(clients.map((c) => [c.id, nomClient(c)])) }));
            setNbClients(clients.length);
            setDemandes(listes.flat().sort((x, y) => y.id - x.id));
          }
        }
      } catch (e) {
        if (!annule) setErreur(e.message);
      } finally {
        if (!annule) setChargement(false);
      }
    }, 300);
    return () => {
      annule = true;
      clearTimeout(minuteur);
    };
  }, [q, statut]);

  // Noms des clients de la liste (hors recherche)
  useEffect(() => {
    const manquants = [...new Set(demandes.map((d) => d.client_id))].filter((id) => !(id in noms));
    if (manquants.length === 0) return;
    Promise.all(
      manquants.map((id) =>
        appeler("/clients/" + id)
          .then((c) => [id, nomClient(c)])
          .catch(() => [id, "Client n° " + id])
      )
    ).then((paires) => setNoms((a) => ({ ...a, ...Object.fromEntries(paires) })));
  }, [demandes]); // eslint-disable-line react-hooks/exhaustive-deps

  return (
    <Cadre titre={utilisateur.role === "decideur" ? "Dossiers" : "Dossiers de crédit"}>
      <input
        className="recherche"
        type="search"
        placeholder="Rechercher un client : nom, prénom, téléphone ou code (2 caractères minimum)"
        value={q}
        onChange={(e) => setQ(e.target.value)}
      />
      <label className="filtre">Statut
        <select value={statut} onChange={(e) => setStatut(e.target.value)}>
          <option value="">Tous</option>
          {Object.entries(LIBELLE_STATUT).map(([cle, libelle]) => (
            <option key={cle} value={cle}>{libelle}</option>
          ))}
        </select>
      </label>
      {erreur && <p className="erreur">{erreur}</p>}
      {chargement ? (
        <p className="discret">Chargement…</p>
      ) : (
        <p className="discret">
          {demandes.length} dossier(s)
          {nbClients !== null && " pour " + nbClients + " client(s) trouvé(s)"}.
          {nbClients === 20 && " Résultats limités : précisez la recherche."}
        </p>
      )}
      <table className="tableau">
        <thead>
          <tr><th>Numéro</th><th>Client</th><th>Montant</th><th>Durée</th><th>Date</th><th>Statut</th><th></th></tr>
        </thead>
        <tbody>
          {demandes.map((d) => (
            <tr key={d.id}>
              <td>{d.numero}</td>
              <td>{noms[d.client_id] || "…"}</td>
              <td>{fcfa(d.montant_demande)}</td>
              <td>{d.duree_mois} mois</td>
              <td>{dateFr(d.date_demande)}</td>
              <td><span className="badge">{LIBELLE_STATUT[d.statut] || d.statut}</span></td>
              <td><Link to={"/dossiers/" + d.id}>Ouvrir</Link></td>
            </tr>
          ))}
        </tbody>
      </table>
    </Cadre>
  );
}
import { useCallback, useEffect, useState } from "react";
import { Link, useParams } from "react-router-dom";
import { appeler } from "./api";
import { useAuth } from "./auth.jsx";
import { dateFr, fcfa } from "./formats.js";
import { LIBELLE_DECISION, LIBELLE_STATUT } from "./listes.js";
import Cadre from "./Cadre.jsx";

const RECOMMANDATION = { accord: "Accord", accord_reduit: "Accord réduit", refus: "Refus" };
const STATUTS_CREDIT = ["decaisse", "en_remboursement", "solde", "contentieux"];

function pct(x) {
  return (Number(x) * 100).toFixed(1).replace(".", ",") + " %";
}

function Facteurs({ facteurs }) {
  if (!facteurs) return null;
  const lignes = Array.isArray(facteurs) ? facteurs.map((x, i) => [i + 1, x]) : Object.entries(facteurs);
  return (
    <ul>
      {lignes.map(([cle, valeur]) => (
        <li key={cle}>
          <strong>{cle}</strong> : {typeof valeur === "object" ? JSON.stringify(valeur) : String(valeur)}
        </li>
      ))}
    </ul>
  );
}

export default function FicheDossier() {
  const { id } = useParams();
  const { utilisateur } = useAuth();
  const gestion = ["agent_credit", "administrateur"].includes(utilisateur.role);
  const directeur = utilisateur.role === "decideur";

  const [d, setD] = useState(null);
  const [client, setClient] = useState(null);
  const [score, setScore] = useState(null);
  const [decisions, setDecisions] = useState([]);
  const [credit, setCredit] = useState(null);
  const [erreur, setErreur] = useState("");
  const [message, setMessage] = useState("");
  const [occupe, setOccupe] = useState(false);
  const [avis, setAvis] = useState("");
  const [dec, setDec] = useState({ decision: "accorder", montant: "", duree: "", commentaire: "" });

  const charger = useCallback(async () => {
    try {
      const demande = await appeler("/demandes/" + id);
      setD(demande);
      appeler("/clients/" + demande.client_id).then(setClient).catch(() => setClient(null));
      appeler("/demandes/" + id + "/score").then(setScore).catch(() => setScore(null));
      appeler("/demandes/" + id + "/decisions").then(setDecisions).catch(() => setDecisions([]));
      appeler("/demandes/" + id + "/credit").then(setCredit).catch(() => setCredit(null));
    } catch (e) {
      setErreur(e.message);
    }
  }, [id]);

  useEffect(() => {
    charger();
  }, [charger]);

  async function agir(chemin, json, succes) {
    setOccupe(true);
    setErreur("");
    setMessage("");
    try {
      await appeler("/demandes/" + id + chemin, { methode: "POST", json });
      setMessage(succes);
      await charger();
    } catch (e) {
      setErreur(e.message);
    } finally {
      setOccupe(false);
    }
  }

  function choisirDecision(valeur) {
    setDec((a) => ({
      ...a,
      decision: valeur,
      montant: a.montant || String((score && score.montant_suggere) || d.montant_demande),
      duree: a.duree || String(d.duree_mois),
    }));
  }

  function envoyerDecision(evenement) {
    evenement.preventDefault();
    const corps = { decision: dec.decision, commentaire: dec.commentaire.trim() || null };
    if (dec.decision === "accorder_modifie") {
      corps.montant_accorde = parseInt(dec.montant, 10);
      corps.duree_accordee = parseInt(dec.duree, 10);
    }
    agir("/decider", corps, "Décision enregistrée.");
  }

  if (!d) {
    return (
      <Cadre titre="Dossier">
        {erreur ? <p className="erreur">{erreur}</p> : <p className="discret">Chargement…</p>}
        <Link to="/dossiers">Retour aux dossiers</Link>
      </Cadre>
    );
  }

  return (
    <Cadre titre={"Dossier " + d.numero}>
      <section className="panneau">
        <h2>Demande <span className="badge">{LIBELLE_STATUT[d.statut] || d.statut}</span></h2>
        <dl className="grille-info">
          <div><dt>Client</dt><dd>
            <Link to={"/clients/" + d.client_id}>{client ? client.prenom + " " + client.nom : "Client n° " + d.client_id}</Link>
          </dd></div>
          <div><dt>Montant demandé</dt><dd>{fcfa(d.montant_demande)}</dd></div>
          <div><dt>Durée</dt><dd>{d.duree_mois} mois</dd></div>
          <div><dt>Objet</dt><dd>{d.objet || "—"}</dd></div>
          <div><dt>Date de la demande</dt><dd>{dateFr(d.date_demande)}</dd></div>
        </dl>
        {d.avis_instruction && (
          <><h3>Avis d'instruction</h3><p>{d.avis_instruction}</p></>
        )}
      </section>

      {score && (
        <section className="panneau">
          <h2>Score du modèle</h2>
          {score.valide === false && (
            <p className="avertissement">
              Ce score n'est plus valide : les données du client ont changé depuis le calcul. Il faut le recalculer.
            </p>
          )}
          <dl className="grille-info">
            <div><dt>Probabilité de défaut</dt><dd>{pct(score.p_defaut)}</dd></div>
            <div><dt>Seuil appliqué</dt><dd>{pct(score.seuil_applique)}</dd></div>
            <div><dt>Recommandation</dt><dd>{RECOMMANDATION[score.recommandation] || score.recommandation}</dd></div>
            <div><dt>Montant suggéré</dt><dd>{score.montant_suggere ? fcfa(score.montant_suggere) : "—"}</dd></div>
            <div><dt>Modèle</dt><dd>{score.version_modele}</dd></div>
          </dl>
          {score.facteurs && (<><h3>Facteurs</h3><Facteurs facteurs={score.facteurs} /></>)}
        </section>
      )}

      {gestion && (
        <section className="panneau">
          <h2>Actions</h2>
          {d.statut === "brouillon" && (
            <button disabled={occupe} onClick={() => agir("/completer", undefined, "Dossier marqué complet.")}>
              Marquer le dossier complet
            </button>
          )}
          {(d.statut === "complet" || d.statut === "score") && (
            <button disabled={occupe} onClick={() => agir("/scorer", undefined, "Score calculé.")}>
              {d.statut === "complet" ? "Calculer le score" : "Recalculer le score"}
            </button>
          )}
          {d.statut === "score" && (
            <form className="formulaire-simple" onSubmit={(e) => { e.preventDefault(); agir("/instruire", { avis: avis.trim() }, "Avis enregistré."); }}>
              <label>Avis d'instruction (10 caractères minimum)
                <textarea rows={4} value={avis} onChange={(e) => setAvis(e.target.value)} required minLength={10} maxLength={2000} />
              </label>
              <button type="submit" disabled={occupe}>Enregistrer l'avis</button>
            </form>
          )}
          {d.statut === "instruit" && (
            <button disabled={occupe} onClick={() => agir("/soumettre", undefined, "Dossier soumis à la décision.")}>
              Soumettre à la décision du directeur
            </button>
          )}
          {["accorde", "accorde_modifie"].includes(d.statut) && (
            <button disabled={occupe} onClick={() => agir("/decaisser", undefined, "Crédit décaissé.")}>
              Décaisser le crédit
            </button>
          )}
          {["refuse", "ajourne", "solde", "decaisse", "en_remboursement", "contentieux", "soumis_decision"].includes(d.statut) && (
            <p className="discret">Aucune action disponible pour ce statut.</p>
          )}
        </section>
      )}

      {directeur && d.statut === "soumis_decision" && (
        <section className="panneau">
          <h2>Décision</h2>
          <p className="discret">
            Si votre décision diffère de la recommandation du modèle, un commentaire est obligatoire.
          </p>
          <form className="formulaire-simple" onSubmit={envoyerDecision}>
            <label>Décision
              <select value={dec.decision} onChange={(e) => choisirDecision(e.target.value)}>
                {Object.entries(LIBELLE_DECISION).map(([cle, libelle]) => (
                  <option key={cle} value={cle}>{libelle}</option>
                ))}
              </select>
            </label>
            {dec.decision === "accorder_modifie" && (
              <>
                <label>Montant accordé (FCFA)
                  <input type="number" min="1" value={dec.montant} required
                    onChange={(e) => setDec({ ...dec, montant: e.target.value })} />
                </label>
                <label>Durée accordée (mois)
                  <input type="number" min="1" max="120" value={dec.duree} required
                    onChange={(e) => setDec({ ...dec, duree: e.target.value })} />
                </label>
              </>
            )}
            <label>Commentaire
              <textarea rows={3} value={dec.commentaire} maxLength={2000}
                onChange={(e) => setDec({ ...dec, commentaire: e.target.value })} />
            </label>
            <button type="submit" disabled={occupe}>Enregistrer la décision</button>
          </form>
        </section>
      )}

      {decisions.length > 0 && (
        <section className="panneau">
          <h2>Historique des décisions</h2>
          <table className="tableau">
            <thead>
              <tr><th>Date</th><th>Décision</th><th>Montant</th><th>Durée</th><th>Modèle</th><th>Concordance</th><th>Commentaire</th></tr>
            </thead>
            <tbody>
              {decisions.map((x) => (
                <tr key={x.id}>
                  <td>{dateFr(x.decide_le.slice(0, 10))}</td>
                  <td>{LIBELLE_DECISION[x.decision] || x.decision}</td>
                  <td>{x.montant_accorde ? fcfa(x.montant_accorde) : "—"}</td>
                  <td>{x.duree_accordee ? x.duree_accordee + " mois" : "—"}</td>
                  <td>{RECOMMANDATION[x.recommandation_modele] || x.recommandation_modele || "—"}</td>
                  <td>{x.concordance === null ? "—" : x.concordance ? "Oui" : "Non"}</td>
                  <td>{x.commentaire || ""}</td>
                </tr>
              ))}
            </tbody>
          </table>
        </section>
      )}

      {credit && STATUTS_CREDIT.includes(d.statut) && (
        <section className="panneau">
          <h2>Crédit</h2>
          <dl className="grille-info">
            <div><dt>Montant accordé</dt><dd>{fcfa(credit.montant_accorde)}</dd></div>
            <div><dt>Durée</dt><dd>{credit.duree_mois} mois</dd></div>
            <div><dt>Taux annuel</dt><dd>{credit.taux_annuel}</dd></div>
            <div><dt>Mensualité</dt><dd>{fcfa(credit.mensualite)}</dd></div>
            <div><dt>Décaissé le</dt><dd>{dateFr(credit.date_decaissement)}</dd></div>
          </dl>
        </section>
      )}

      {message && <p className="succes">{message}</p>}
      {erreur && <p className="erreur">{erreur}</p>}
      <p><Link to="/dossiers">Retour aux dossiers</Link></p>
    </Cadre>
  );
}
