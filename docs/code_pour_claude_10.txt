import { useEffect, useState } from "react";
import { Link, useNavigate, useSearchParams } from "react-router-dom";
import { appeler } from "./api";
import { fcfa } from "./formats.js";
import Cadre from "./Cadre.jsx";

export default function NouvelleDemande() {
  const [params] = useSearchParams();
  const clientId = params.get("client_id");
  const navigate = useNavigate();
  const [client, setClient] = useState(null);
  const [produits, setProduits] = useState([]);
  const [produitId, setProduitId] = useState("");
  const [montant, setMontant] = useState("");
  const [duree, setDuree] = useState("");
  const [objet, setObjet] = useState("");
  const [erreur, setErreur] = useState("");
  const [envoi, setEnvoi] = useState(false);

  useEffect(() => {
    if (!clientId) {
      setErreur("Aucun client choisi : ouvrez d'abord la fiche d'un client.");
      return;
    }
    appeler("/clients/" + clientId).then(setClient).catch((e) => setErreur(e.message));
    appeler("/produits")
      .then((liste) => {
        setProduits(liste);
        if (liste.length > 0) setProduitId(String(liste[0].id));
      })
      .catch((e) => setErreur(e.message));
  }, [clientId]);

  const produit = produits.find((p) => String(p.id) === produitId);

  async function creer(evenement) {
    evenement.preventDefault();
    setErreur("");
    setEnvoi(true);
    try {
      const demande = await appeler("/demandes", {
        methode: "POST",
        json: {
          client_id: parseInt(clientId, 10),
          produit_id: parseInt(produitId, 10),
          montant_demande: parseInt(montant, 10),
          duree_mois: parseInt(duree, 10),
          objet: objet.trim() || null,
        },
      });
      navigate("/dossiers/" + demande.id);
    } catch (e) {
      setErreur(e.message);
      setEnvoi(false);
    }
  }

  return (
    <Cadre titre="Nouvelle demande de crédit">
      {client && (
        <p>Client : <strong>{client.prenom} {client.nom}</strong> ({client.code_client})</p>
      )}
      <form className="formulaire" onSubmit={creer}>
        <label>Produit
          <select value={produitId} onChange={(e) => setProduitId(e.target.value)} required>
            {produits.map((p) => (
              <option key={p.id} value={p.id}>{p.libelle}</option>
            ))}
          </select>
        </label>
        <label>
          {"Montant demandé (FCFA)" + (produit ? " : de " + fcfa(produit.montant_min) + " à " + fcfa(produit.montant_max) : "")}
          <input type="number" value={montant} onChange={(e) => setMontant(e.target.value)} required
            min={produit ? produit.montant_min : 1} max={produit ? produit.montant_max : undefined} />
        </label>
        <label>
          {"Durée (mois)" + (produit ? " : de " + produit.duree_min_mois + " à " + produit.duree_max_mois : "")}
          <input type="number" value={duree} onChange={(e) => setDuree(e.target.value)} required
            min={produit ? produit.duree_min_mois : 1} max={produit ? produit.duree_max_mois : 120} />
        </label>
        <label>Objet du crédit
          <input value={objet} onChange={(e) => setObjet(e.target.value)} maxLength={200} />
        </label>
        <div className="actions">
          {erreur && <p className="erreur">{erreur}</p>}
          <button type="submit" disabled={envoi || !client}>{envoi ? "Création…" : "Créer la demande"}</button>
          {clientId && <Link to={"/clients/" + clientId}>Annuler</Link>}
        </div>
      </form>
    </Cadre>
  );
}
import { useEffect, useState } from "react";
import { appeler } from "./api";
import { LIBELLE_CATEGORIE } from "./listes.js";
import { invaliderMetiers } from "./metiers.js";
import Cadre from "./Cadre.jsx";

export default function Metiers() {
  const [aClasser, setAClasser] = useState([]);
  const [categories, setCategories] = useState([]);
  const [choix, setChoix] = useState({});
  const [erreur, setErreur] = useState("");
  const [message, setMessage] = useState("");
  const [chargement, setChargement] = useState(true);

  useEffect(() => {
    Promise.all([appeler("/professions/a-classer"), appeler("/professions/categories")])
      .then(([liste, cats]) => {
        setAClasser(liste);
        setCategories(cats);
      })
      .catch((e) => setErreur(e.message))
      .finally(() => setChargement(false));
  }, []);

  async function classer(metier) {
    setErreur("");
    setMessage("");
    try {
      await appeler("/professions/" + metier.code, {
        methode: "PUT",
        json: { categorie_modele: choix[metier.code] },
      });
      invaliderMetiers();
      setAClasser((liste) => liste.filter((m) => m.code !== metier.code));
      setMessage(
        "« " + metier.libelle + " » est classé. Les scores en cours des " + metier.nb_clients +
          " client(s) concerné(s) doivent être recalculés."
      );
    } catch (e) {
      setErreur(e.message);
    }
  }

  return (
    <Cadre titre="Métiers à classer">
      <p className="discret">
        Un métier saisi librement n'a pas de catégorie pour le modèle. Tant qu'il n'est pas classé,
        le score des clients concernés utilise « Non renseigné ». Choisissez la catégorie la plus proche.
      </p>
      {erreur && <p className="erreur">{erreur}</p>}
      {message && <p className="succes">{message}</p>}
      {chargement && <p className="discret">Chargement…</p>}
      {!chargement && aClasser.length === 0 && <p>Aucun métier à classer.</p>}
      {aClasser.length > 0 && (
        <table className="tableau">
          <thead>
            <tr><th>Métier</th><th>Clients</th><th>Catégorie du modèle</th><th></th></tr>
          </thead>
          <tbody>
            {aClasser.map((m) => (
              <tr key={m.code}>
                <td>{m.libelle}</td>
                <td>{m.nb_clients}</td>
                <td>
                  <select value={choix[m.code] || ""} onChange={(e) => setChoix({ ...choix, [m.code]: e.target.value })}>
                    <option value="">Choisir…</option>
                    {categories.map((c) => (
                      <option key={c} value={c}>{(LIBELLE_CATEGORIE[c] || c) + " (" + c + ")"}</option>
                    ))}
                  </select>
                </td>
                <td><button disabled={!choix[m.code]} onClick={() => classer(m)}>Classer</button></td>
              </tr>
            ))}
          </tbody>
        </table>
      )}
    </Cadre>
  );
}
import { useId } from "react";
import { useMetiers } from "./metiers.js";

// Champ libre avec suggestions : on peut choisir un métier de la liste ou écrire le sien.
export default function ChampMetier({ valeur, onChange, metiers }) {
  const identifiant = useId();
  const deLaListe = useMetiers();
  const liste = metiers || deLaListe;
  return (
    <>
      <input
        list={identifiant}
        value={valeur}
        onChange={(e) => onChange(e.target.value)}
        maxLength={60}
        autoComplete="off"
        placeholder="Choisir dans la liste ou écrire un métier"
      />
      <datalist id={identifiant}>
        {liste.map((m) => (
          <option key={m.code} value={m.libelle} />
        ))}
      </datalist>
    </>
  );
}
