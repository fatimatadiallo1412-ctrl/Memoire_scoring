from sqlalchemy import Boolean, DateTime, ForeignKey, String, text
from sqlalchemy.orm import Mapped, mapped_column

from app.models.ops import SCHEMA, Base


class Metier(Base):
    """Référentiel des métiers : chaque métier est rattaché à une catégorie connue du modèle.

    categorie_modele vide = métier saisi librement, « à classer » : le client est alors scoré
    avec « Non renseigné » jusqu'à ce qu'un responsable choisisse une catégorie.
    """
    __tablename__ = "metier"
    __table_args__ = ({"schema": SCHEMA},)

    id: Mapped[int] = mapped_column(primary_key=True)
    code: Mapped[str] = mapped_column(String(60), unique=True)
    libelle: Mapped[str] = mapped_column(String(100))
    categorie_modele: Mapped[str | None] = mapped_column(String(40))
    valide: Mapped[bool] = mapped_column(Boolean, default=False, server_default=text("false"))
    actif: Mapped[bool] = mapped_column(Boolean, default=True, server_default=text("true"))
    cree_par: Mapped[int | None] = mapped_column(ForeignKey(f"{SCHEMA}.utilisateur.id"))
    cree_le: Mapped[datetime] = mapped_column(DateTime(timezone=True), server_default=text("now()"))
from fastapi import APIRouter, Depends
from pydantic import BaseModel, ConfigDict
from sqlalchemy import select
from sqlalchemy.orm import Session

from app.db import get_session
from app.deps import exiger_roles
from app.models.enums import Role
from app.models.ops import Agence, ProduitCredit

router = APIRouter(tags=["référentiels"])

PERSONNEL = (Role.AGENT_CREDIT, Role.DECIDEUR, Role.ADMINISTRATEUR)


class AgenceOut(BaseModel):
    model_config = ConfigDict(from_attributes=True)

    id: int
    code: str
    nom: str
    zone: str


class ProduitOut(BaseModel):
    model_config = ConfigDict(from_attributes=True)

    id: int
    code: str
    libelle: str
    montant_min: int
    montant_max: int
    duree_min_mois: int
    duree_max_mois: int
    taux_annuel: float | None = None


@router.get("/agences", response_model=list[AgenceOut])
def lister_agences(
    session: Session = Depends(get_session),
    _=Depends(exiger_roles(*PERSONNEL)),
):
    """Agences actives, pour les listes déroulantes de l'interface."""
    requete = select(Agence).where(Agence.actif.is_(True)).order_by(Agence.nom)
    return session.scalars(requete).all()


@router.get("/produits", response_model=list[ProduitOut])
def lister_produits(
    session: Session = Depends(get_session),
    _=Depends(exiger_roles(*PERSONNEL)),
):
    """Produits de crédit (montants et durées autorisés), pour le formulaire de demande."""
    requete = select(ProduitCredit).order_by(ProduitCredit.libelle)
    if hasattr(ProduitCredit, "actif"):
        requete = requete.where(ProduitCredit.actif.is_(True))
    return session.scalars(requete).all()
